* หากเว้นว่างไว้ ระบบจะบันทึกและดึงข้อมูลจาก Local Storage ของเบราว์เซอร์อัตโนมัติ

โค้ด Google Apps Script (คัดลอกไปวางใน Google Sheet > ส่วนขยาย > Apps Script): 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["OrderID", "Timestamp", "CustomerName", "Phone", "Address", "SKU001_Qty", "SKU002_Qty", "TotalPrice", "Payment", "Status"]);
  }
  
  sheet.appendRow([
    data.orderId,
    data.timestamp,
    data.name,
    "'" + data.phone,
    data.address,
    data.sku1,
    data.sku2,
    data.total,
    data.payment,
    data.status || "รอดำเนินการ"
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var orders = [];
  for (var i = 1; i < rows.length; i++) {
    orders.push({
      orderId: rows[i][0],
      timestamp: rows[i][1],
      name: rows[i][2],
      phone: String(rows[i][3]).replace(/^'/, ''),
      address: rows[i][4],
      sku1: Number(rows[i][5]) || 0,
      sku2: Number(rows[i][6]) || 0,
      total: Number(rows[i][7]) || 0,
      payment: rows[i][8],
      status: rows[i][9] || "รอดำเนินการ"
    });
  }
  return ContentService.createTextOutput(JSON.stringify(orders)).setMimeType(ContentService.MimeType.JSON);
}
